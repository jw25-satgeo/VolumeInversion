def hilbert2phase(x):
    y = -1j * torch.fft.rfft(x, dim=0); y[0] = 0;
    y = torch.fft.irfft(y, dim=0) #hilbert transform
    y = torch.angle(torch.complex(x, y)) # get phase
    y[1:] += torch.cumsum(y[1:] < y[:-1] - 1.9*torch.pi, dim=0) * 2*torch.pi #unwrap phase
    return y
