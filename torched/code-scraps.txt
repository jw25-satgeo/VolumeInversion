from CSIM import copula_sim
def OGPF(qpol1, qpol2, channels=8, ps=None, fmp_chunk=256, g_kern_std=(1,5,5), device='cuda', tensor=True, p_disp=True):
    '''
    Adapted OGPF : for each patch, determine the most similar scattering metric.
    '''
    H, W = qpol1.shape[1], qpol1.shape[2]
    if (ps is None): #ensures a minimum 64x64 patch size
        ps = 2**max([i for i in range(6, min(H,W).bit_length()) if (i==6 or (H%(2**i)==0 and W%(2**i)==0))])
        ps = [ps,ps]; print('generated patch size:',ps)
    if (ps[0] != ps[1]): raise ValueError(f'Patch {ps} is not square.')
    if (2**((ps[0]).bit_length()-1)!=ps[0] or 2**((ps[1]).bit_length()-1)!=ps[1]):raise ValueError(f'Patch {ps} is not 2^x.')
    
    FMP = torch.zeros((H,W,6,2), device=device, dtype=torch.float16)
    for i in tqdm.trange(H//fmp_chunk + 1 if(H%fmp_chunk > 0) else H//fmp_chunk, desc='FMP decomp chunks', disable=not p_disp):
        i1, i2 = fmp_chunk*i, fmp_chunk*(i+1)
        FMP[i1:i2,:,:,0] = FreemanPlus(qpol1[:,i1:i2], device=device)
        FMP[i1:i2,:,:,1] = FreemanPlus(qpol2[:,i1:i2], device=device)

    #H,W,C,2 to 2,H,W,C to Hc,Wc to Hc,Wc,band
    scores = torch.stack([copula_sim(FMP.permute(3,0,1,2)[...,i,None]) for i in range(FMP.shape[2])], dim=-1)
    scores = torch.argmin(scores, dim=2)[None,None,...].float() #Hc,Wc,band to 1,1,Hc,Wc
    scores = F.interpolate(scores, size=(H,W), mode='nearest').permute(2,3,0,1).expand(-1,-1,-1,2).int() #1,1,Hc,Wc to H,W,1,2
    FMP = FMP.gather(2, scores).permute(3,2,0,1) # H,W,1,2 to 2,1,H,W (B,C,H,W) - so dim0 spans both im1 and im2
    
    sobel_x = torch.tensor([[-1,0,1],[-2,0,2],[-1,0,1]],device=device,dtype=FMP.dtype)[None,None,...]
    sobel_y = torch.tensor([[-1,-2,-1],[0,0,0],[1,2,1]],device=device,dtype=FMP.dtype)[None,None,...]
    gx = F.conv2d(FMP, sobel_x, padding=1, groups=1)
    gy = F.conv2d(FMP, sobel_y, padding=1, groups=1)
    
    theta = torch.linspace(0,np.pi,channels,device=device,dtype=FMP.dtype)[None,:,None,None]
    OGPF = torch.abs(torch.cos(theta) * gx + torch.sin(theta) * gy) #2,C,H,W from 1,C,1,1 x 2,1,H,W

    #Gaussian filter over channel, H, W
    g_std = torch.tensor(g_kern_std, device=device, dtype=OGPF.dtype)
    k = torch.ceil(6 * g_std).int() #length 6 std, or radius 3 std
    tmp = torch.tensor(OGPF.shape[-3:], device=device).to(k.dtype)//3; k[k > tmp] = tmp[k > tmp]
    k = torch.stack(torch.meshgrid(*[torch.arange(k[i],device=device)-k[i]//2 for i in [0,1,2]],indexing='ij'),dim=0)
    k = torch.exp(-0.5*(k**2 / g_std[:,None,None,None]**2)) / (((8*torch.pi**3)**0.5)*torch.prod(g_std))
    k = torch.sum(k, dim=0)[None,None,...] #1,1,z,x,y
    #pad = (k.shape[2]//2,k.shape[2]//2, k.shape[3]//2,k.shape[3]//2, k.shape[4]//2,k.shape[4]//2)
    pad = tuple([k.shape[i//2]//2 for i in range(6)]) #print(OGPF.shape, k.shape, pad)
    OGPF = F.conv3d(F.pad(OGPF.unsqueeze(1), pad, mode='circular'), k) #2,1,C,H,W circular conv 1,1,x,y,z
    OGPF = OGPF.squeeze(1)

    OGPF = OGPF.permute(0,2,3,1) #2,C,H,W to 2,H,W,C
    if (tensor): return OGPF
    return OGPF[0].moveaxis(-1,0).detach().cpu().numpy(), OGPF[1].moveaxis(-1,0).detach.cpu().numpy()
